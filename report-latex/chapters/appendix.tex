\section{Useful Commands}
\begin{itemize}
  \item Run backend API: \texttt{uvicorn backend.app:app --reload}
  \item Run backend tests: \texttt{pytest tests -q}
  \item Run frontend dev server: \texttt{cd frontend \&\& npm run dev}
  \item Run frontend tests: \texttt{cd frontend \&\& npm run test}
\end{itemize}
